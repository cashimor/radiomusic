cmake_minimum_required(VERSION 3.20)
project(Radiomusic VERSION 0.3 LANGUAGES CXX)
if(MSVC)
  add_compile_options(/utf-8 /EHsc)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(music_core src/music.cpp src/features.cpp src/crossover.cpp src/mixer.cpp src/library.cpp)
set_target_properties(music_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(music_core PUBLIC src)
if(MSVC)
  target_compile_options(music_core PRIVATE /W4 /permissive-)
endif()
if(WIN32)
  add_executable(Radiomusic WIN32 src/main.cpp src/engine.cpp src/analysis_worker.cpp)
  target_compile_definitions(Radiomusic PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(Radiomusic SYSTEM PRIVATE vendor)
  target_link_libraries(Radiomusic PRIVATE music_core winhttp winmm comctl32 shell32)
  if(MSVC)
    target_compile_options(Radiomusic PRIVATE /W4 /permissive-)
  endif()
endif()
if(ANDROID)
  add_library(RadioNative SHARED android/app/src/main/cpp/native_player.cpp)
  target_include_directories(RadioNative PRIVATE vendor)
  target_link_libraries(RadioNative PRIVATE music_core aaudio log android)
endif()
enable_testing()
if(NOT ANDROID)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE music_core)
  add_test(NAME core_tests COMMAND core_tests)
endif()
